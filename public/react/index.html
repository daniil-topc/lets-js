Компоненты, пропсы, состояние и хуки. Короткие объяснения, код и примеры из настоящих интерфейсов, которые можно потрогать прямо на странице.

Начать с компонентов Все главы
Counter.jsx
// Твой первый компонент
import { useState } from 'react';

function Counter() {
  const [count, setCount] = useState(0);

  return (
    <button onClick={() => setCount(count + 1)}>
      Нажато: {count}
    </button>
  );
}

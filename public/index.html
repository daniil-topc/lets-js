hello.js
// Твоя первая программа
const name = 'новичок';
let level = 0;

function learn(topic) {
  level++;
  return `Изучено: ${topic} ✓`;
}

learn('переменные');
learn('функции');

console.log(`Привет, ${name}!`);
console.log(`Твой уровень: ${level}`); // 2

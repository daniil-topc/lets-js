Асинхронность

Обычно JavaScript выполняет код по порядку, строку за строкой. Но некоторые дела занимают время: таймер должен досчитать, сервер должен ответить. Если бы страница замирала на эти секунды, пользоваться ей было бы невозможно. Асинхронный код ждёт в фоне, а страница продолжает работать.

setTimeout()

setTimeout работает как будильник. Ты говоришь, что сделать и через сколько, и JavaScript выполнит это один раз, когда время выйдет. Время пишется в миллисекундах: 1000 миллисекунд равны одной секунде.

//         что сделать            через сколько мс
setTimeout(() => {
  console.log('Прошло 2 секунды');
}, 2000);

// Таймер можно отменить, пока он не сработал
const timerId = setTimeout(() => {
  console.log('Этого никто не увидит');
}, 5000);

clearTimeout(timerId);

Код, который стоит после setTimeout, таймера не ждёт и выполняется сразу. Функция внутри сработает позже, когда выйдет время.

Реальный кейс

Уведомление, которое исчезает само

Плашка «Товар добавлен в корзину» появляется сразу после клика и через 3 секунды исчезает сама. Убирает её setTimeout.

const button = document.querySelector('.add-to-cart');
const toast = document.querySelector('.toast');
let timerId;

button.addEventListener('click', () => {
  toast.classList.add('is-visible');

  clearTimeout(timerId); // если нажали ещё раз — отсчёт заново
  timerId = setTimeout(() => {
    toast.classList.remove('is-visible');
  }, 3000);
});

Реальный кейс

Поиск, который не дёргает сервер на каждую букву

Если отправлять запрос на сервер после каждой буквы, слово «Москва» даст шесть запросов, и пять из них лишние. Есть приём под названием debounce: каждая новая буква отменяет прошлый таймер и заводит новый. Поиск запускается, когда человек перестал печатать на полсекунды.

const input = document.querySelector('#search');
let timerId;

input.addEventListener('input', () => {
  clearTimeout(timerId);        // отменяем прошлый запуск
  timerId = setTimeout(() => {
    search(input.value);        // ищем через 500 мс тишины
  }, 500);
});

Введи первые буквы города

Поисков выполнено: 0. Печатай быстро, и число почти не вырастет.

setInterval()

setInterval похож на setTimeout, только срабатывает снова и снова через равные промежутки времени. Так тикают часы на сайте и идёт обратный отсчёт. Сам он не остановится, поэтому, когда повторы больше не нужны, вызови clearInterval.

let seconds = 0;

const timerId = setInterval(() => {
  seconds++;
  console.log(`Прошло секунд: ${seconds}`);

  if (seconds === 5) {
    clearInterval(timerId); // обязательно останавливаем
  }
}, 1000);
Реальный кейс

«Отправить код повторно через 10 с»

При входе по SMS кнопка «Отправить код» блокируется на несколько секунд, а надпись рядом показывает, сколько осталось ждать. Надпись раз в секунду обновляет setInterval.

function startCountdown(button, label) {
  let seconds = 10;
  button.disabled = true;

  const timerId = setInterval(() => {
    seconds--;
    label.textContent = `Повторить можно через ${seconds} с`;

    if (seconds === 0) {
      clearInterval(timerId);
      button.disabled = false;
      label.textContent = 'Можно отправить код ещё раз';
    }
  }, 1000);
}

Промисы

В кафе тебе дают номерок: заказ ещё не готов, но его принесут. Промис в JavaScript работает как этот номерок. Результата пока нет, а объект, который его обещает, уже у тебя. Дальше возможны два исхода: заказ принесли или на кухне что-то пошло не так. На первый случай подписываются через .then(), на второй через .catch().

// Функция, которая «ждёт» указанное время
function wait(ms) {
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

wait(1000)
  .then(() => {
    console.log('Прошла секунда');
  })
  .catch((error) => {
    console.log('Что-то пошло не так', error);
  });

async / await

Цепочки из .then() быстро становятся длинными. Слова async и await позволяют записать то же самое проще. await ставит функцию на паузу, пока промис не выполнится, и код читается сверху вниз, как обычный.

async function showMessages() {
  console.log('Загрузка…');
  await wait(1000);          // ждём секунду
  console.log('Готово!');
}

showMessages();

Кнопка запускает функцию с await wait(2000). Пока она ждёт, страница не зависает: попробуй прокрутить её или выделить текст.

await работает только внутри функции, перед которой написано async. Ошибки ловят блоком try...catch.

fetch()

fetch отправляет запрос на сервер и возвращает промис с ответом. Благодаря ему сайты подгружают ленту, комментарии или курс валют без перезагрузки страницы. Ответ обычно приходит в формате JSON, и метод response.json() превращает его в объект.

async function loadUser() {
  try {
    const response = await fetch('/api/user');

    if (!response.ok) {
      throw new Error(`Ошибка: ${response.status}`);
    }

    const user = await response.json(); // JSON → объект
    console.log(user.name);
  } catch (error) {
    console.log('Не удалось загрузить данные');
  }
}
Реальный кейс

Загрузка данных с сервера по кнопке

Кнопка ниже отправляет настоящий запрос серверу этого сайта. Сервер написан на Node.js и по адресу /api/tip отдаёт случайный совет в формате JSON. В коде три шага: показать «Загрузка…», получить ответ и обработать ошибку, если сеть подвела.

const button = document.querySelector('button');
const output = document.querySelector('.output');

button.addEventListener('click', async () => {
  button.disabled = true;
  output.textContent = 'Загрузка…';

  try {
    const response = await fetch('/api/tip');
    if (!response.ok) throw new Error(response.status);

    const tip = await response.json();
    output.textContent = `Совет №${tip.id}: ${tip.text}`;
  } catch (error) {
    output.textContent = 'Не удалось загрузить совет';
  } finally {
    button.disabled = false; // выполнится в любом случае
  }
});
